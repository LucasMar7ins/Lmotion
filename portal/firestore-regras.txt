Regras de segurança do Firestore — Portal Lmotion

Como aplicar:
1. No Firebase Console, entre no projeto "Lmotion".
2. Vá em Firestore Database > aba "Regras".
3. Apague o conteúdo atual e cole o texto abaixo.
4. Clique em "Publicar".

Essas regras garantem que:
- Cada cliente só lê os próprios dados (nunca de outro cliente).
- Só o admin pode criar clientes, entregar vídeos, adicionar materiais e publicar tutoriais.
- O cliente só pode escrever nos campos de feedback do próprio vídeo, enviar pauta e enviar dúvida.
- O cliente pode registrar notificações de atividade (só criar, nunca ler); só o admin lê e marca como lida.
- O cliente pode apagar os próprios materiais enviados (caso tenha subido algo por engano).
- O portal registra automaticamente o último acesso do cliente (só esse campo, sem acesso a mais nada).
- O admin também pode adicionar materiais em nome de um cliente (ex: quando ele manda pelo WhatsApp).
- Pagamentos são de uso exclusivo do admin — o cliente nunca lê nem escreve nada nessa coleção.
- Métricas de Instagram, TikTok e YouTube (crescimento de seguidores/inscritos) são de uso exclusivo do admin.

---

rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function logado() {
      return request.auth != null;
    }

    function isAdmin() {
      return logado() &&
        exists(/databases/$(database)/documents/clientes/$(request.auth.uid)) &&
        get(/databases/$(database)/documents/clientes/$(request.auth.uid)).data.isAdmin == true;
    }

    // Cada cliente lê o próprio cadastro; admin lê e escreve tudo.
    // O cliente também pode confirmar presença / pedir remarcação nas próprias gravações
    // (lista "gravacoes"), salvar a observação geral e preencher o "briefing"
    // (aba Sobre o negócio).
    match /clientes/{uid} {
      allow read: if logado() && (request.auth.uid == uid || isAdmin());
      allow create, delete: if isAdmin();
      allow update: if isAdmin() || (
        logado() &&
        request.auth.uid == uid &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['gravacoes', 'observacaoGravacao', 'ultimoAcesso', 'briefing'])
      );
    }

    // Vídeos / entregas
    match /entregas/{id} {
      allow read: if logado() && (resource.data.clienteId == request.auth.uid || isAdmin());
      allow create: if isAdmin();
      allow update: if isAdmin() || (
        logado() &&
        resource.data.clienteId == request.auth.uid &&
        request.resource.data.diff(resource.data).affectedKeys().hasOnly(['feedback'])
      );
      allow delete: if isAdmin();
    }

    // Pautas / sugestões de tema
    match /pautas/{id} {
      allow read: if logado() && (resource.data.clienteId == request.auth.uid || isAdmin());
      allow create: if logado() && request.resource.data.clienteId == request.auth.uid;
      allow update, delete: if isAdmin();
    }

    // Materiais de marca (o cliente envia e pode apagar os próprios; admin também pode)
    match /materiais/{id} {
      allow read: if logado() && (resource.data.clienteId == request.auth.uid || isAdmin());
      allow create: if isAdmin() || (logado() && request.resource.data.clienteId == request.auth.uid);
      allow delete: if isAdmin() || (logado() && resource.data.clienteId == request.auth.uid);
      allow update: if isAdmin();
    }

    // Dúvidas
    match /duvidas/{id} {
      allow read: if logado() && (resource.data.clienteId == request.auth.uid || isAdmin());
      allow create: if logado() && request.resource.data.clienteId == request.auth.uid;
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }

    // Notificações de atividade (feed do admin com as ações dos clientes)
    match /notificacoes/{id} {
      allow read, update, delete: if isAdmin();
      allow create: if logado() && request.resource.data.clienteId == request.auth.uid;
    }

    // Pagamentos — controle interno, só o admin acessa
    match /pagamentos/{id} {
      allow read, write: if isAdmin();
    }

    // Instagram — crescimento de seguidores, controle interno, só o admin acessa
    match /instagramMetricas/{id} {
      allow read, write: if isAdmin();
    }

    // TikTok — crescimento de seguidores, controle interno, só o admin acessa
    match /tiktokMetricas/{id} {
      allow read, write: if isAdmin();
    }

    // YouTube — crescimento de inscritos, controle interno, só o admin acessa
    match /youtubeMetricas/{id} {
      allow read, write: if isAdmin();
    }

    // Tutoriais — visíveis a qualquer cliente logado, só o admin publica
    match /tutoriais/{id} {
      allow read: if logado();
      allow write: if isAdmin();
    }

    // Anotações do calendário do Dashboard — cada admin só vê e escreve as próprias
    match /notasDashboard/{uid}/dias/{id} {
      allow read, write: if logado() && request.auth.uid == uid && isAdmin();
    }
  }
}
